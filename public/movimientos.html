const API = "http://localhost:3000/api/movimientos"; let todosLosMovimientos = []; // ====================== ESTADÍSTICAS ====================== async function cargarEstadisticas() { const res = await fetch(API); todosLosMovimientos = await res.json(); const entradas = todosLosMovimientos.filter(m => m.tipo).length; const salidas = todosLosMovimientos.filter(m => !m.tipo).length; const totalHoy = todosLosMovimientos.filter(m => m.fecha === new Date().toISOString().split('T')[0]).length; document.getElementById('stats').innerHTML = `
Entradas Totales

${entradas}

Salidas Totales

${salidas}

Movimientos Hoy

${totalHoy}

Total Movimientos

${todosLosMovimientos.length}

`; } // ====================== FILTROS Y TABLA ====================== function aplicarFiltros() { const producto = document.getElementById('filtroProducto').value; const tipo = document.getElementById('filtroTipo').value; const desde = document.getElementById('filtroDesde').value; const hasta = document.getElementById('filtroHasta').value; const busqueda = document.getElementById('busqueda').value.toLowerCase(); let filtrados = todosLosMovimientos.filter(m => { const coincideProducto = !producto || m.id_producto == producto; const coincideTipo = !tipo || m.tipo.toString() === tipo; const coincideFecha = (!desde || m.fecha >= desde) && (!hasta || m.fecha <= hasta); const coincideBusqueda = m.nombre_producto.toLowerCase().includes(busqueda); return coincideProducto && coincideTipo && coincideFecha && coincideBusqueda; }); renderizarTabla(filtrados); } function renderizarTabla(data) { const tbody = document.querySelector('#tablaMovimientos tbody'); tbody.innerHTML = ''; data.forEach(m => { const fila = document.createElement('tr'); fila.innerHTML = ` ${m.fecha} ${m.tipo ? 'Entrada' : 'Salida'} ${m.nombre_producto} ${m.nombre_almacen} ${m.cantidad} `; tbody.appendChild(fila); }); } // ====================== EXPORTAR ====================== function exportarCSV() { let csv = "Fecha,Tipo,Producto,Almacén,Cantidad\n"; todosLosMovimientos.forEach(m => { csv += `${m.fecha},${m.tipo ? 'Entrada' : 'Salida'},${m.nombre_producto},${m.nombre_almacen},${m.cantidad}\n`; }); const blob = new Blob([csv], { type: 'text/csv' }); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = `Movimientos_Agromundo_${new Date().toISOString().slice(0,10)}.csv`; a.click(); } function exportarPDF() { window.print(); // Abre la impresión (puedes guardar como PDF) } // ====================== INICIALIZACIÓN ====================== async function init() { await cargarEstadisticas(); cargarEstadisticas(); // segunda vez para llenar filtros renderizarTabla(todosLosMovimientos); // Llenar filtro de productos const productosUnicos = [...new Set(todosLosMovimientos.map(m => m.id_producto))]; const select = document.getElementById('filtroProducto'); productosUnicos.forEach(id => { const nombre = todosLosMovimientos.find(m => m.id_producto === id).nombre_producto; const opt = document.createElement('option'); opt.value = id; opt.textContent = nombre; select.appendChild(opt); }); } init();
